// ============================================================
// ARDUINO MEGA 2560
// BRAILLE INPUT -> SIMPAN KALIMAT -> ENTER PLAYBACK
// OLED SSD1306 + 6 SOLENOIDS
// ============================================================
//
// BRAILLE:
//
//       1   4
//       2   5
//       3   6
//
// BUTTON 1 -> DOT 1
// BUTTON 2 -> DOT 2
// BUTTON 3 -> DOT 3
// BUTTON 4 -> DOT 4
// BUTTON 5 -> DOT 5
// BUTTON 6 -> DOT 6
//
// PIN SOLENOID:
// DOT 1 -> D2
// DOT 2 -> D3
// DOT 3 -> D4
// DOT 4 -> D5
// DOT 5 -> D6
// DOT 6 -> D7
//
// PIN BUTTON:
// BUTTON 1 -> D8
// BUTTON 2 -> D9
// BUTTON 3 -> D10
// BUTTON 4 -> D11
// BUTTON 5 -> D12
// BUTTON 6 -> D13
//
// ENTER -> A0
//
// OLED:
// SDA -> D20
// SCL -> D21
// ADDRESS -> 0x3C
//
// CARA KERJA:
//
// 1. Tekan button untuk membentuk SATU huruf.
// 2. Setelah tidak ada button lagi selama 900 ms,
//    pola dianggap selesai dan huruf disimpan.
// 3. Selama mengetik, SOLENOID TIDAK BERGERAK.
// 4. OLED menunjukkan huruf calon + kalimat tersimpan.
// 5. ENTER 1x = PLAY seluruh kalimat.
// 6. ENTER 2x cepat = RESET.
// ============================================================

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>


// ============================================================
// OLED
// ============================================================

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
#define OLED_ADDRESS 0x3C

Adafruit_SSD1306 display(
  SCREEN_WIDTH,
  SCREEN_HEIGHT,
  &Wire,
  OLED_RESET
);


// ============================================================
// PIN SOLENOID
// URUTAN FISIK TIDAK DIUBAH
// ============================================================

const byte solenoidPins[6] = {
  2,  // Dot 1
  3,  // Dot 2
  4,  // Dot 3
  5,  // Dot 4
  6,  // Dot 5
  7   // Dot 6
};


// ============================================================
// PIN BUTTON
// ============================================================

const byte buttonPins[6] = {
  8,   // Button 1 -> Dot 1
  9,   // Button 2 -> Dot 2
  10,  // Button 3 -> Dot 3
  11,  // Button 4 -> Dot 4
  12,  // Button 5 -> Dot 5
  13   // Button 6 -> Dot 6
};


// ============================================================
// ENTER
// ============================================================

const byte ENTER_PIN = A0;


// ============================================================
// BRAILLE TABLE
// [Dot1, Dot2, Dot3, Dot4, Dot5, Dot6]
// ============================================================

const byte braille[26][6] = {

  {1,0,0,0,0,0},  // A
  {1,1,0,0,0,0},  // B
  {1,0,0,1,0,0},  // C
  {1,0,0,1,1,0},  // D
  {1,0,0,0,1,0},  // E

  {1,1,0,1,0,0},  // F
  {1,1,0,1,1,0},  // G
  {1,1,0,0,1,0},  // H
  {0,1,0,1,0,0},  // I
  {0,1,0,1,1,0},  // J

  {1,0,1,0,0,0},  // K
  {1,1,1,0,0,0},  // L
  {1,0,1,1,0,0},  // M
  {1,0,1,1,1,0},  // N
  {1,0,1,0,1,0},  // O

  {1,1,1,1,0,0},  // P
  {1,1,1,1,1,0},  // Q
  {1,1,1,0,1,0},  // R
  {0,1,1,1,0,0},  // S
  {0,1,1,1,1,0},  // T

  {1,0,1,0,0,1},  // U
  {1,1,1,0,0,1},  // V
  {0,1,0,1,1,1},  // W
  {1,0,1,1,0,1},  // X
  {1,0,1,1,1,1},  // Y
  {1,0,1,0,1,1}   // Z
};


// ============================================================
// INPUT SATU HURUF
// ============================================================

byte inputPattern[6] = {
  0, 0, 0,
  0, 0, 0
};


// ============================================================
// KALIMAT YANG SUDAH TERSIMPAN
// ============================================================

const byte MAX_TEXT = 80;

char sentence[MAX_TEXT + 1] = "";


// ============================================================
// BUTTON DEBOUNCE
// INPUT_PULLUP
//
// HIGH = tidak ditekan
// LOW  = ditekan
// ============================================================

bool rawButton[6];
bool stableButton[6];

unsigned long buttonChangeTime[6];

const unsigned long BUTTON_DEBOUNCE = 60;


// ============================================================
// INPUT HURUF
// ============================================================

bool inputActive = false;

unsigned long lastInputTime = 0;

// Waktu antar tombol untuk membentuk SATU huruf
const unsigned long INPUT_TIMEOUT = 900;


// ============================================================
// ENTER DEBOUNCE
// ============================================================

bool rawEnter = HIGH;
bool stableEnter = HIGH;

unsigned long enterChangeTime = 0;

const unsigned long ENTER_DEBOUNCE = 100;


// ============================================================
// DOUBLE ENTER
//
// ENTER 1x = PLAY
// ENTER 2x cepat = RESET
// ============================================================

byte enterCount = 0;

unsigned long firstEnterTime = 0;

const unsigned long DOUBLE_ENTER_WINDOW = 600;


// ============================================================
// PLAYBACK
// ============================================================

bool playbackActive = false;
bool letterActive = false;

int playbackIndex = 0;

unsigned long playbackTimer = 0;

// Lama satu huruf di solenoid
const unsigned long LETTER_ON_TIME = 1000;

// Jeda antarhuruf
const unsigned long LETTER_GAP_TIME = 350;


// ============================================================
// CLEAR INPUT
// ============================================================

void clearInput() {

  for (byte i = 0; i < 6; i++) {
    inputPattern[i] = 0;
  }

  inputActive = false;
}


// ============================================================
// ALL SOLENOID OFF
// ============================================================

void allSolenoidsOff() {

  for (byte i = 0; i < 6; i++) {
    digitalWrite(solenoidPins[i], LOW);
  }
}


// ============================================================
// CEK POLA KOSONG
// ============================================================

bool isEmptyPattern(const byte pattern[6]) {

  for (byte i = 0; i < 6; i++) {

    if (pattern[i] != 0) {
      return false;
    }
  }

  return true;
}


// ============================================================
// POLA BRAILLE -> HURUF
// ============================================================

char patternToLetter(const byte pattern[6]) {

  for (byte letterIndex = 0; letterIndex < 26; letterIndex++) {

    bool match = true;

    for (byte dot = 0; dot < 6; dot++) {

      if (pattern[dot] != braille[letterIndex][dot]) {
        match = false;
        break;
      }
    }

    if (match) {
      return char('A' + letterIndex);
    }
  }

  return '?';
}


// ============================================================
// HURUF -> POLA BRAILLE
// ============================================================

void letterToPattern(char letter, byte pattern[6]) {

  // Spasi
  if (letter == ' ') {

    for (byte i = 0; i < 6; i++) {
      pattern[i] = 0;
    }

    return;
  }

  if (letter < 'A' || letter > 'Z') {

    for (byte i = 0; i < 6; i++) {
      pattern[i] = 0;
    }

    return;
  }

  byte index = letter - 'A';

  for (byte i = 0; i < 6; i++) {
    pattern[i] = braille[index][i];
  }
}


// ============================================================
// TAMBAH KARAKTER KE KALIMAT
// ============================================================

void appendCharacter(char c) {

  byte length = strlen(sentence);

  if (length < MAX_TEXT) {

    sentence[length] = c;
    sentence[length + 1] = '\0';

  } else {

    // Geser ke kiri jika penuh
    for (byte i = 0; i < MAX_TEXT - 1; i++) {
      sentence[i] = sentence[i + 1];
    }

    sentence[MAX_TEXT - 1] = c;
    sentence[MAX_TEXT] = '\0';
  }
}


// ============================================================
// APPLY SATU POLA KE SEMUA SOLENOID
//
// SATU-SATUNYA FUNGSI YANG MENGENDALIKAN OUTPUT SOLENOID
//
// Dot 1 -> D2
// Dot 2 -> D3
// Dot 3 -> D4
// Dot 4 -> D5
// Dot 5 -> D6
// Dot 6 -> D7
// ============================================================

void applyPatternToSolenoids(const byte pattern[6]) {

  for (byte dot = 0; dot < 6; dot++) {

    digitalWrite(
      solenoidPins[dot],
      pattern[dot] ? HIGH : LOW
    );
  }
}


// ============================================================
// GAMBAR BRAILLE DI OLED
// ============================================================

void drawBraille(
  const byte pattern[6],
  int x,
  int y
) {

  const int radius = 5;

  int leftX = x;
  int rightX = x + 28;

  int y1 = y;
  int y2 = y + 15;
  int y3 = y + 30;


  // Dot 1
  if (pattern[0])
    display.fillCircle(leftX, y1, radius, SSD1306_WHITE);
  else
    display.drawCircle(leftX, y1, radius, SSD1306_WHITE);


  // Dot 2
  if (pattern[1])
    display.fillCircle(leftX, y2, radius, SSD1306_WHITE);
  else
    display.drawCircle(leftX, y2, radius, SSD1306_WHITE);


  // Dot 3
  if (pattern[2])
    display.fillCircle(leftX, y3, radius, SSD1306_WHITE);
  else
    display.drawCircle(leftX, y3, radius, SSD1306_WHITE);


  // Dot 4
  if (pattern[3])
    display.fillCircle(rightX, y1, radius, SSD1306_WHITE);
  else
    display.drawCircle(rightX, y1, radius, SSD1306_WHITE);


  // Dot 5
  if (pattern[4])
    display.fillCircle(rightX, y2, radius, SSD1306_WHITE);
  else
    display.drawCircle(rightX, y2, radius, SSD1306_WHITE);


  // Dot 6
  if (pattern[5])
    display.fillCircle(rightX, y3, radius, SSD1306_WHITE);
  else
    display.drawCircle(rightX, y3, radius, SSD1306_WHITE);
}


// ============================================================
// OLED MODE INPUT
// ============================================================

void showInputDisplay() {

  display.clearDisplay();

  display.setTextColor(SSD1306_WHITE);


  // ------------------------------------------
  // CALON HURUF
  // ------------------------------------------

  char currentLetter =
    patternToLetter(inputPattern);


  display.setTextSize(1);

  display.setCursor(0, 0);
  display.print("CALON: ");


  display.setTextSize(2);

  display.setCursor(55, 0);

  if (!inputActive) {

    display.print("-");

  } else if (currentLetter == '?') {

    display.print("?");

  } else {

    display.print(currentLetter);
  }


  // ------------------------------------------
  // POLA
  // ------------------------------------------

  drawBraille(
    inputPattern,
    25,
    24
  );


  // ------------------------------------------
  // KALIMAT
  // ------------------------------------------

  display.setTextSize(1);

  display.setCursor(65, 28);
  display.print("TEXT:");


  byte length = strlen(sentence);

  byte start = 0;

  if (length > 10) {
    start = length - 10;
  }


  display.setCursor(65, 40);

  for (byte i = start; i < length; i++) {
    display.print(sentence[i]);
  }


  display.setCursor(65, 56);
  display.print("ENTER=PLAY");


  display.display();
}


// ============================================================
// OLED MODE PLAYBACK
// ============================================================

void showPlaybackDisplay(char letter) {

  display.clearDisplay();

  display.setTextColor(SSD1306_WHITE);


  display.setTextSize(1);

  display.setCursor(0, 0);
  display.print("PLAY:");


  display.setTextSize(2);

  display.setCursor(50, 0);

  if (letter == ' ') {
    display.print("SP");
  } else {
    display.print(letter);
  }


  // Gunakan pola yang SAMA dengan solenoid
  byte pattern[6];

  letterToPattern(
    letter,
    pattern
  );


  drawBraille(
    pattern,
    25,
    24
  );


  // Kalimat
  display.setTextSize(1);

  display.setCursor(65, 28);
  display.print("TEXT:");


  byte length = strlen(sentence);

  byte start = 0;

  if (length > 10) {
    start = length - 10;
  }


  display.setCursor(65, 40);

  for (byte i = start; i < length; i++) {
    display.print(sentence[i]);
  }


  display.display();
}


// ============================================================
// DAFTARKAN SATU DOT
// ============================================================

void registerDot(byte dot) {

  // HANYA DOT INI YANG BERUBAH
  inputPattern[dot] =
    !inputPattern[dot];


  inputActive = true;

  lastInputTime =
    millis();


  // OLED berubah
  // Solenoid TIDAK berubah
  showInputDisplay();
}


// ============================================================
// SIMPAN SATU HURUF
// ============================================================

void saveCurrentCharacter() {

  if (!inputActive) {
    return;
  }


  char letter =
    patternToLetter(inputPattern);


  if (letter != '?') {

    appendCharacter(letter);
  }


  clearInput();

  showInputDisplay();
}


// ============================================================
// PROSES TIMEOUT INPUT
// ============================================================

void processInputTimeout() {

  if (!inputActive) {
    return;
  }


  if (
    millis() - lastInputTime >=
    INPUT_TIMEOUT
  ) {

    saveCurrentCharacter();
  }
}


// ============================================================
// START PLAYBACK
// ============================================================

void startPlayback() {

  byte length =
    strlen(sentence);


  if (length == 0) {
    return;
  }


  playbackActive = true;
  playbackIndex = 0;
  letterActive = false;
  playbackTimer = millis();


  allSolenoidsOff();
}


// ============================================================
// UPDATE PLAYBACK
// ============================================================

void updatePlayback() {

  if (!playbackActive) {
    return;
  }


  byte length =
    strlen(sentence);


  // ------------------------------------------
  // SELESAI
  // ------------------------------------------

  if (playbackIndex >= length) {

    playbackActive = false;
    letterActive = false;

    allSolenoidsOff();

    showInputDisplay();

    return;
  }


  // ------------------------------------------
  // MULAI HURUF BARU
  // ------------------------------------------

  if (!letterActive) {

    char letter =
      sentence[playbackIndex];


    byte pattern[6];

    letterToPattern(
      letter,
      pattern
    );


    // ======================================================
    // POLA YANG SAMA KEPADA:
    //
    // OLED
    // SOLENOID
    //
    // ======================================================

    applyPatternToSolenoids(
      pattern
    );


    showPlaybackDisplay(
      letter
    );


    letterActive = true;
    playbackTimer = millis();

    return;
  }


  // ------------------------------------------
  // HURUF SEDANG AKTIF
  // ------------------------------------------

  unsigned long elapsed =
    millis() - playbackTimer;


  if (elapsed >= LETTER_ON_TIME) {

    allSolenoidsOff();
  }


  if (
    elapsed >=
    LETTER_ON_TIME +
    LETTER_GAP_TIME
  ) {

    playbackIndex++;
    letterActive = false;
    playbackTimer = millis();
  }
}


// ============================================================
// RESET SEMUA
// ============================================================

void resetSystem() {

  sentence[0] = '\0';

  clearInput();

  inputActive = false;

  playbackActive = false;
  playbackIndex = 0;
  letterActive = false;

  enterCount = 0;

  allSolenoidsOff();

  showInputDisplay();
}


// ============================================================
// RAW + STABLE BUTTON READING
// ============================================================

void readButtons() {

  unsigned long now =
    millis();


  for (byte i = 0; i < 6; i++) {

    bool raw =
      digitalRead(
        buttonPins[i]
      );


    // ----------------------------------------
    // RAW BERUBAH
    // ----------------------------------------

    if (
      raw != rawButton[i]
    ) {

      rawButton[i] =
        raw;

      buttonChangeTime[i] =
        now;
    }


    // ----------------------------------------
    // SUDAH STABIL
    // ----------------------------------------

    if (
      now - buttonChangeTime[i] >=
      BUTTON_DEBOUNCE
    ) {

      if (
        stableButton[i] != raw
      ) {

        bool previous =
          stableButton[i];


        stableButton[i] =
          raw;


        // ----------------------------------
        // HANYA HIGH -> LOW
        // ----------------------------------

        if (
          previous == HIGH &&
          stableButton[i] == LOW
        ) {

          // Saat playback aktif,
          // button diabaikan
          if (!playbackActive) {

            registerDot(i);
          }
        }
      }
    }
  }
}


// ============================================================
// ENTER READING
// ============================================================

void readEnter() {

  unsigned long now =
    millis();


  bool raw =
    digitalRead(
      ENTER_PIN
    );


  // ----------------------------------------
  // RAW BERUBAH
  // ----------------------------------------

  if (
    raw != rawEnter
  ) {

    rawEnter = raw;
    enterChangeTime = now;
  }


  // ----------------------------------------
  // STABIL
  // ----------------------------------------

  if (
    now - enterChangeTime >=
    ENTER_DEBOUNCE
  ) {

    if (
      stableEnter != raw
    ) {

      bool previous =
        stableEnter;


      stableEnter =
        raw;


      // ------------------------------------
      // ENTER DITEKAN
      // ------------------------------------

      if (
        previous == HIGH &&
        stableEnter == LOW
      ) {

        handleEnter();
      }
    }
  }
}


// ============================================================
// HANDLE ENTER
//
// ENTER 1x = PLAY
// ENTER 2x cepat = RESET
//
// ============================================================

void handleEnter() {

  unsigned long now =
    millis();


  // Saat playback, ENTER kedua bisa reset
  if (playbackActive) {

    if (enterCount == 0) {

      enterCount = 1;
      firstEnterTime = now;

    } else {

      if (
        now - firstEnterTime <=
        DOUBLE_ENTER_WINDOW
      ) {

        resetSystem();

      } else {

        enterCount = 1;
        firstEnterTime = now;
      }
    }

    return;
  }


  // ----------------------------------------
  // ENTER PERTAMA
  // ----------------------------------------

  if (enterCount == 0) {

    enterCount = 1;

    firstEnterTime =
      now;

    return;
  }


  // ----------------------------------------
  // ENTER KEDUA
  // ----------------------------------------

  if (
    now - firstEnterTime <=
    DOUBLE_ENTER_WINDOW
  ) {

    // RESET
    resetSystem();

  } else {

    // Jadikan ENTER pertama lagi
    enterCount = 1;

    firstEnterTime =
      now;
  }
}


// ============================================================
// PROSES ENTER SETELAH WINDOW
// ============================================================

void processEnter() {

  if (
    enterCount == 0
  ) {
    return;
  }


  // Tunggu window double-enter selesai
  if (
    millis() - firstEnterTime >=
    DOUBLE_ENTER_WINDOW
  ) {

    // ENTER 1x -> PLAY
    enterCount = 0;

    if (!playbackActive) {
      startPlayback();
    }
  }
}


// ============================================================
// SETUP
// ============================================================

void setup() {

  // ------------------------------------------
  // SOLENOID
  // ------------------------------------------

  for (byte i = 0; i < 6; i++) {

    pinMode(
      solenoidPins[i],
      OUTPUT
    );

    digitalWrite(
      solenoidPins[i],
      LOW
    );
  }


  allSolenoidsOff();


  // ------------------------------------------
  // BUTTON
  // ------------------------------------------
  //
  // Tombol:
  // D8  -- BUTTON -- GND
  // D9  -- BUTTON -- GND
  // ...
  //
  // INPUT_PULLUP
  //
  // ------------------------------------------

  for (byte i = 0; i < 6; i++) {

    pinMode(
      buttonPins[i],
      INPUT_PULLUP
    );


    rawButton[i] =
      digitalRead(
        buttonPins[i]
      );


    stableButton[i] =
      rawButton[i];


    buttonChangeTime[i] =
      millis();
  }


  // ------------------------------------------
  // ENTER
  // ------------------------------------------

  pinMode(
    ENTER_PIN,
    INPUT_PULLUP
  );


  rawEnter =
    digitalRead(
      ENTER_PIN
    );


  stableEnter =
    rawEnter;


  enterChangeTime =
    millis();


  // ------------------------------------------
  // OLED
  // ------------------------------------------

  Wire.begin();


  if (
    !display.begin(
      SSD1306_SWITCHCAPVCC,
      OLED_ADDRESS
    )
  ) {

    allSolenoidsOff();

    while (true) {
      delay(100);
    }
  }


  // ------------------------------------------
  // START
  // ------------------------------------------

  display.clearDisplay();

  display.setTextColor(
    SSD1306_WHITE
  );

  display.setTextSize(2);

  display.setCursor(15, 8);
  display.print("BRAILLE");

  display.setTextSize(1);

  display.setCursor(18, 40);
  display.print("MEGA 2560");

  display.display();

  delay(1200);


  sentence[0] =
    '\0';

  clearInput();

  allSolenoidsOff();

  showInputDisplay();
}


// ============================================================
// LOOP
// ============================================================

void loop() {

  // 1. Baca tombol
  readButtons();


  // 2. Simpan huruf setelah 900 ms tidak ada tombol baru
  processInputTimeout();


  // 3. Baca ENTER
  readEnter();


  // 4. Proses ENTER 1x / 2x
  processEnter();


  // 5. Playback kalimat
  updatePlayback();

}
